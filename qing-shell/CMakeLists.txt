cmake_minimum_required(VERSION 3.24)
project(qing_shell VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "qing-shell requires the MSVC Windows SDK toolchain")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Build the shell component for x64 or ARM64 Explorer")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_compile_options(/W4 /WX /EHsc /utf-8 /guard:cf)
add_link_options(/DYNAMICBASE /NXCOMPAT /guard:cf)
add_library(qing-shell SHARED src/commands.cpp src/qing-shell.def)
target_link_libraries(qing-shell PRIVATE ole32 shell32 shlwapi advapi32 uuid)
add_executable(qing-shell-host WIN32 src/host.cpp)
target_link_libraries(qing-shell-host PRIVATE shell32)
enable_testing()
add_executable(qing-shell-args-tests tests/args.cpp)
target_link_libraries(qing-shell-args-tests PRIVATE shell32)
add_executable(qing-shell-argv-fixture WIN32 tests/argv-fixture.cpp)
target_link_libraries(qing-shell-argv-fixture PRIVATE shell32)
add_test(NAME command_line_roundtrip COMMAND qing-shell-args-tests $<TARGET_FILE:qing-shell-argv-fixture>)
add_executable(qing-shell-com-tests tests/com.cpp)
target_link_libraries(qing-shell-com-tests PRIVATE ole32 shell32 uuid)
add_test(NAME isolated_com_contract COMMAND qing-shell-com-tests $<TARGET_FILE:qing-shell>)
add_executable(qing-shell-selection-tests tests/selection.cpp)
target_link_libraries(qing-shell-selection-tests PRIVATE ole32 shell32 shlwapi advapi32 uuid)
add_test(NAME filesystem_selection COMMAND qing-shell-selection-tests)
install(TARGETS qing-shell qing-shell-host RUNTIME DESTINATION . LIBRARY DESTINATION .)

# Invoked only by the disposable Windows runner, after the desktop build.
add_executable(qing-shell-classic-forwarding tests/classic-menu-forwarding.cpp)
target_link_libraries(qing-shell-classic-forwarding PRIVATE ole32 shell32 user32 uuid)
